# TMNT (MAME set 'tmnt', World 4 Players, version X) - Z80 sound program symbols (963e20.g13)
#
# Addresses and names only: no game data lives in this file.
# Format:  ADDRESS  KIND  NAME  [...]   [# comment: how it was found]
#   func     a function entry (one nothing calls is decompiled too: code the entry points never reach)
#   loop     the main loop's top, a function entry: where the program waits for the board (runtime/src/sound_cpu.c
#            runs the main loop from there while the board's time is ahead)
#   code     a code label (a place the code jumps into the middle of a routine: set_channel_handler)
#   jumps    NAME COUNT [first=K]: a table of COUNT 16-bit code addresses, which the code indexes from
#            ADDRESS; with first=K it never reads the first K entries (the table starts K entries on)
#   data     NAME TYPE[COUNT] [TARGET]: COUNT values of TYPE (u8, u16, or a record: struct NAME SIZE rom);
#            TARGET, what each u16 points at (a record, or the elements a pointer walks)
#   codedata NAME TYPE[COUNT]: code the program also reads as data (the bytes stay code)
#   fill     COUNT: bytes nothing reads (none: the program's own fill is the 0xFF after its last track)
# Addresses are the Z80's: 0000-7FFF the program ROM, 8000-87FF the RAM, 9000 sres (bit 2 the title music
# sample), A000 the 68000's sound command (the latch), B000-B00D the K007232, C000-C001 the YM2151, D000 the
# uPD7759's sample number, E000 its start line, F000 its busy line.
#
# The analysis (tools/core/z80/analysis.py, game_profile's SOUND_ settings) starts at the reset (0000) and
# the interrupt (0038) and finds the code without traces, following what the registers hold into the jumps
# through them (z80/values.py): the main loop runs each channel through the handler its record keeps at +2
# (run_channel_handler, JP (HL) at 05CF), which the code stores there (set_channel_handler and the like),
# and the music's note hook at +28 (09FC). The jumps lines give the tables' lengths, which the code does not.
# The data lines describe the rest of the program for tools/sound_coverage.py; tools/sound_data_walk.py
# walks the music tracks the song table points at.
#
# Names for the decompiled C (tools/decompile_sound.py), as symbols.txt has them (disassemble.load_names):
#   struct NAME SIZE rom, field STRUCT OFFSET NAME TYPE[N] [TARGET]: a record of the program ROM and its
#   fields; elements NAME TYPE: values a pointer walks; param FUNCTION REGISTER NAME STRUCT: what a function
#   takes in a register (the sound coverage takes every constant the analysis sees there as one);
#   ADDRESS io NAME TYPE: a register of the board on the Z80's bus; ADDRESS var NAME[COUNT] TYPE: a variable in
#   the RAM; ADDRESS table NAME TYPE: a table the C reads by name (the game loads the asset piece of that name);
#   struct NAME SIZE, array STRUCT FIRST COUNT, field: records in the RAM; scoped STRUCT FAMILY OFFSET NAME TYPE:
#   a field as the functions whose names start with FAMILY use it (the music's fields are the record's own, the
#   FM and PCM effects name the same bytes their own way: fm_effect, pcm_effect and one family for an effect that
#   uses a byte otherwise, which takes the rest from the family its name starts with); param FUNCTION REGISTER
#   NAME [STRUCT[.FAMILY]]: what a function takes or hands back in a register; local FUNCTION WHERE REGISTER NAME:
#   the variable the instructions at WHERE write (decompile_sound.py --variables lists the sites); comment
#   FUNCTION TEXT: what the C says above it; constant FUNCTION VALUE NAME: a number by name in one function.
#   Every name has its evidence above it: the instruction of build/sound/listing.s that shows it.

# ======== the driver ========
#
# What the program is (read from the listing): the 68000 writes a sound command to the latch and raises the
# Z80's IRQ; sound_command (0038) takes it at once: 00 resets the sound, FF silences it, FE fades the music
# out, 01-41 start an FM effect, 60-78 a song, 7D and 7E start and stop the title music (the 512 KB sample
# MAME's tmnt.cpp plays through sres bit 2), 80-93 a PCM effect (the K007232), A0-C3 a speech (the
# uPD7759), DF stops the speech. The main loop waits for the YM2151's timer, then runs 15 channel records of
# 40h bytes (8000-83BF), each through the handler it keeps at +2: 8000 the FM effects that come first (YM2151
# channel 0), 8040-81C0 the music's 7 FM channels (1-7) and 8200, 8240 its 2 PCM channels (the K007232's A
# and B), 8280 and 82C0 the PCM effects (A, B), 8300 the speech, 8340 and 8380 the FM effects that take the
# music's channels 6 and 7. A record's +0 is its chip channel (bit 3: the K007232, bit 0 its channel B), +1
# the sound it plays, +F its priority.

0017  func  check_rom                   # the ROM test: adds up the program's 32 KB (0000-7FFF) and compares the sum with 783C, then sound_main (the game has no ROM to add up: runtime/sound_replaced.txt)

00FE  code  stop_speech                 # command DF: sres 0 then 2 (the uPD7759 reset by bit 1, the title music off by bit 2), the speech channel idle
0121  code  speech_fallback             # a speech whose priority is below the one playing: one of priority 12 plays PCM effect 87 instead; the test for 13 (0128 "cp $13") comes after 0123 "ld a,$87", so it compares the effect's number and PCM effect 8A never plays

# ---- what the C says above each function of the driver ----
comment sound_reset The reset: tests the RAM with four patterns and then the ROM; a failure beeps and starts again.
comment check_rom Adds up the program ROM and compares the sum with the one it should have, then starts the program (the game's board puts its own in place of it: there is no ROM to add up).
comment sound_command The interrupt: takes the 68000's command and starts what it asks for: a reset, silence, the music's fade-out, the title music, an FM effect, a song, a PCM effect or a speech.
comment self_test_failed A failed test: a beep on the K007232, then the reset again.
comment stop_title_music Stops the title music, and with it a speech (they share the line that resets the speech chip).
comment title_music_timer Counts the main loop's passes while the title music plays: after 13 rounds of 250 it stops it.
comment start_speech Starts a speech on the speech channel when its priority is not below the one speaking (the speech chip reset first); one of priority 12 that cannot plays PCM effect 87 instead (the test meant for 13 compares the effect's number, so it never holds). Command DF stops the speech.
comment stop_speech Stops the speech: the speech chip reset, the title music off, the speech channel free, the music back to its attenuation.
comment speech_play The speech channel's first pass: the speech's number to the speech chip and a pulse on its start line; the music turned down while it speaks.
comment speech_wait Waits for the speech chip to finish; then frees the channel and turns the music back up.
comment start_pcm_effect Starts a PCM effect on the PCM effects' channel B (effect 86 on channel A) when its priority is not below the one playing there; an effect that plays already starts again.
comment start_fm_effect Starts an FM effect: one of priority 20 or more on the first FM effects' channel; effects 0A and 21 on the music's channel 6, and one that plays already on its own channel; the others on whichever of the three has the lowest priority, when theirs is not below it.
comment lower_priority_channel Of the channel found so far and this one, the one with the lower priority.
comment reset_sound Command 00 and the start: the speech, the music's fade and attenuation and the FM effects' flags cleared, then silence.
comment silence Command FF: no LFO on the 8 FM channels and every channel stopped (the music's PCM channels only freed while an effect holds them).

struct  fm_voice 19 rom                 # a YM2151 voice: load_fm_voice 05F9-06B1 writes it a byte at a time ("inc hl; ld c,(hl)"), operators M1, C1, M2, C2
field   fm_voice 0  connection u8       # register 20: both outputs, feedback, algorithm (into +5)
field   fm_voice 1  m1_detune_multiple u8   # 40
field   fm_voice 2  m1_total_level u8   # into +6 (written with the volume: set_fm_volume, 60)
field   fm_voice 3  m1_attack u8        # 80: key scaling, attack rate
field   fm_voice 4  m1_decay_1 u8       # A0: AMS enable, first decay rate
field   fm_voice 5  m1_decay_2 u8       # C0: detune 2, second decay rate
field   fm_voice 6  m1_release u8       # E0: first decay level, release rate (into +A)
field   fm_voice 7  c1_detune_multiple u8   # 50
field   fm_voice 8  c1_total_level u8   # into +7 (70)
field   fm_voice 9  c1_attack u8        # 90
field   fm_voice A  c1_decay_1 u8       # B0
field   fm_voice B  c1_decay_2 u8       # D0
field   fm_voice C  c1_release u8       # F0 (into +B)
field   fm_voice D  m2_detune_multiple u8   # 48
field   fm_voice E  m2_total_level u8   # into +8 (68)
field   fm_voice F  m2_attack u8        # 88
field   fm_voice 10 m2_decay_1 u8       # A8
field   fm_voice 11 m2_decay_2 u8       # C8
field   fm_voice 12 m2_release u8       # E8 (into +C)
field   fm_voice 13 c2_detune_multiple u8   # 58
field   fm_voice 14 c2_total_level u8   # into +9 (78)
field   fm_voice 15 c2_attack u8        # 98
field   fm_voice 16 c2_decay_1 u8       # B8
field   fm_voice 17 c2_decay_2 u8       # D8
field   fm_voice 18 c2_release u8       # F8 (into +D)

struct  pcm_instrument 4 rom            # play_pcm_note 0C3B-0C56: "ld e,(hl); inc hl; ld d,(hl); inc hl; ld c,(hl); inc hl" then set_pcm_start, "ld a,(hl)"
field   pcm_instrument 0 start u16      # the sample's start in the K007232's ROM (bits 0-15)
field   pcm_instrument 2 bank u8        # its bit 16
field   pcm_instrument 3 level u8       # its volume before the attenuations (into +3C)

struct  pcm_drum 6 rom                  # as pcm_instrument, and the pitch (0C6B/0C6D "ld e,(hl); inc hl; ld d,(hl)", set_pcm_pitch)
field   pcm_drum 0 start u16
field   pcm_drum 2 bank u8
field   pcm_drum 3 level u8
field   pcm_drum 4 pitch u16
# 0C3A "add hl,de": HL at the note's record, a drum's of pcm_drums by the pitch (0C37 "ld hl,$100C") or the
# instrument's of pcm_instruments by +38 (0C29 "ld hl,$1002"); 0C3B-0C40 read its start and bank, 0C56 its level,
# 0C6B-0C6D a drum's pitch. The code reads both kinds as one: an instrument's record is a drum's without the pitch
local   read_track 0C3A hl pcm_sound pcm_drum

struct  song 12 rom                     # a song: the tracks of its 9 channels
field   song 0 tracks u16[9] music_track    # FM channels 1-7, then the K007232's A and B (music_track_start 08BE/08C3 into +10)
elements music_track u8                 # a track's commands, which read_track_command reads a byte at a time (tools/sound_data_walk.py walks them)
elements track_command_handler u16      # an entry of track_commands_e0 or track_commands_f0: a command's handler
# 0D47 "add hl,bc": HL at the command's entry, of track_commands_e0 (0D37 "ld hl,$0D4D") or of track_commands_f0
# (0D41 "ld hl,$0D6D"), the two ways meeting here; 0D48-0D4B read it, 0D4C "jp (hl)" goes there
local   read_track 0D47 hl command_entry track_command_handler
